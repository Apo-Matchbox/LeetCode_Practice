# (Linked List Cycle II) [https://leetcode.com/problems/linked-list-cycle-ii/description/]
# language: C++

* 今回はset

```cpp
#include <iostream>
#include <set>

struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

class Solution {
 public:
    ListNode *detectCycle(ListNode *head) {
        std::set<ListNode*> reached;
        ListNode* node = head;
        while (node) {
            if (reached.contains(node)) {
                return node;
            }
            reached.insert(node);
            node = node->next;
        }
        return nullptr;
    }
};
```
